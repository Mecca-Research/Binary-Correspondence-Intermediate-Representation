define i32 @select_after_overflow(i32 %x) {
entry:
  %sum = add nsw i32 %x, 2147483647
  %is_positive = icmp sgt i32 %sum, 0
  br i1 %is_positive, label %positive, label %nonpositive

positive:
  ret i32 1

nonpositive:
  ret i32 0
}
