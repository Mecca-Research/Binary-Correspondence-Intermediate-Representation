; Adversarial BCIR fixture: this assembles and verifies, but branching on poison

define i32 @poison_feeds_branch(i32 %x) {
entry:
  %cmp = icmp sgt i32 %x, poison
  br i1 %cmp, label %taken, label %fallthrough

taken:
  ret i32 1

fallthrough:
  ret i32 0
}
