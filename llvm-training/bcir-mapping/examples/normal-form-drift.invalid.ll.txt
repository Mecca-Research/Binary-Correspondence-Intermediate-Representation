; llvm-training-invalid-kind: semantic-only
; BCIR semantic failure: two source registers are silently coalesced, the
; address space is collapsed, byte strides are implicit, and a required runtime
; wrapper is bypassed. LLVM accepts this module; a BCIR verifier must reject it.

target datalayout = "e-p:64:64-p1:64:64-i64:64-n8:16:32:64-S128"

define void @normal_form_drift(ptr addrspace(1) %base, i64 %index, i32 %r7, i32 %r8) {
entry:
  %collapsed = add i32 %r7, %r8, !bcir.reg !0
  %generic = addrspacecast ptr addrspace(1) %base to ptr
  %element = getelementptr i32, ptr %generic, i64 %index
  store i32 %collapsed, ptr %element, align 4
  ret void
}

!0 = !{!"r7+r8", i32 2}
